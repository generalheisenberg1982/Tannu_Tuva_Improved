TAN_initialize_toka_support_gui = {
	hidden_effect = {
		set_variable = {
			var = TAN.Kurul_Support
			value = 0
		}
		set_variable = {
			var = TAN.Public_Support
			value = 0
		}
		TAN_refresh_toka_support_gui = yes
	}
}

TAN_refresh_toka_support_gui = {
	hidden_effect = {
		TAN_clamp_toka_support_values = yes
		TAN_update_little_kurul_support_frame = yes
		TAN_update_public_support_frame = yes
	}
}

TAN_clamp_toka_support_values = {
	hidden_effect = {
		if = {
			limit = { check_variable = { TAN.Kurul_Support < 0 } }
			set_variable = {
				var = TAN.Kurul_Support
				value = 0
			}
		}
		if = {
			limit = { check_variable = { TAN.Kurul_Support > 100 } }
			set_variable = {
				var = TAN.Kurul_Support
				value = 100
			}
		}
		if = {
			limit = { check_variable = { TAN.Public_Support < 0 } }
			set_variable = {
				var = TAN.Public_Support
				value = 0
			}
		}
		if = {
			limit = { check_variable = { TAN.Public_Support > 100 } }
			set_variable = {
				var = TAN.Public_Support
				value = 100
			}
		}
	}
}
